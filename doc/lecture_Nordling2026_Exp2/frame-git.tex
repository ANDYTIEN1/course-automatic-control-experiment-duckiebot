% frame-git.tex — Git for the group repository (22:00–23:30)
% Beginner level (user request 2026-11-12: students do not know git). One
% member creates the group's copy of the course student repository and adds the
% others as collaborators. Commands are standard git; nothing repository-specific.
\begin{frame}[fragile]{Git: share your work through one group repository}{One member creates it and adds the others as collaborators}
  \scriptsize
  \begin{columns}[T]
    \column{0.58\textwidth}
\begin{verbatim}
git clone <URL of your group repository>
cd <repository folder>
git pull
git status
git add data/raw/db17-r5-left-m9/pi/<log>
git commit -m "Add Pi logs of E2, motor m9"
git push
\end{verbatim}
    \vspace{-6pt}
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item \texttt{clone}: copy the repository to your PC, once.
      \item \texttt{pull}: fetch the group's latest work; \alert{always before you start}.
      \item \texttt{status}: list what you changed.
      \item \texttt{add}: choose the files to save, \alert{each by name}.
      \item \texttt{commit}: save them with a message that says what and why.
      \item \texttt{push}: send your commits to the group.
    \end{itemize}
    \column{0.39\textwidth}
    \textbf{Where things go}
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item data: \texttt{data/raw/}
      \item code: \texttt{scripts/}
      \item report: its own folder in \texttt{doc/}
    \end{itemize}
    \textbf{Push refused?} Someone pushed first: run \texttt{git pull}, fix any lines marked \texttt{<<<<<<<}, \texttt{add}, \texttt{commit}, \texttt{push}.

    \vspace{2pt}
    \begin{alertblock}{\scriptsize Never commit}
      \scriptsize
      videos; \texttt{build/}; \texttt{.venv/}; \texttt{data/interim/}, \texttt{data/processed/}.
      Never \texttt{git add .}
    \end{alertblock}
  \end{columns}
\end{frame}
